Sample SQL Optimization Query:

WITH SegmentRevenue AS (
  SELECT 
    c.customer_segment,
    COUNT(DISTINCT t.transaction_id) AS total_orders,
    SUM(t.order_value) AS total_revenue,
    AVG(t.profit_margin) AS avg_margin
  FROM transactions t
  JOIN customers c ON t.customer_id = c.customer_id
  GROUP BY c.customer_segment
)
SELECT *,
  ROUND((total_revenue / SUM(total_revenue) OVER()) * 100, 2) AS pct_revenue_contribution
FROM SegmentRevenue
ORDER BY total_revenue DESC;
